    Product_ID INT PRIMARY KEY,
    Product_Name VARCHAR(50) NOT NULL,
    Category_ID INT,
    Price DECIMAL(10,2),
    MOQ INT NOT NULL,
    Listing_Type VARCHAR(20),
    Key_Features VARCHAR(100),
    Units_Sold INT,
    Stock INT,
    FOREIGN KEY (Category_ID) REFERENCES Category(Category_ID)
);
